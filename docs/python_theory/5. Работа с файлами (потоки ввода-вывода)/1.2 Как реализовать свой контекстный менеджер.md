
Код довольно простой, надо только написать класс у которого реализованы два магических метода `__enter()__` и `__exit()__`:

```python

class MyFile():
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
 
    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file
 
    def __exit__(self, type, value, traceback):
        self.file.close()


# теперь мы можем использовать ключевое слово with с объектами нашего класса
with MyFile('test.txt', 'w') as myfile:
    myfile.write('Hello, world!')

```
